    <script>
    /**
     * BONG-KUN'S TRIP SHOOTER
     * 擬人化ボング「ボングくん」のサイケデリック縦スクロールシューティング
     */

    const HIGHSCORE_STORAGE_KEY = 'bongkun_adventure_highscores_v1';

    function loadHighScores() {
        try {
            const raw = localStorage.getItem(HIGHSCORE_STORAGE_KEY);
            if (!raw) return [];
            const arr = JSON.parse(raw);
            if (!Array.isArray(arr)) return [];
            return arr
                .filter(entry => entry && typeof entry.score === 'number' && Number.isFinite(entry.score))
                .map(entry => ({
                    score: Math.max(0, Math.floor(entry.score)),
                    stage: Math.max(1, Math.floor(Number.isFinite(entry.stage) ? entry.stage : 1)),
                    date: typeof entry.date === 'string' && entry.date ? entry.date : '----/--/-- --:--'
                }))
                .sort((a, b) => b.score - a.score)
                .slice(0, 10);
        } catch (_) {
            return [];
        }
    }

    function recordHighScore(score, stage) {
        try {
            const s = Math.max(0, Math.floor(Number.isFinite(score) ? score : 0));
            if (s <= 0) return loadHighScores();
            const st = Math.max(1, Math.floor(Number.isFinite(stage) ? stage : 1));
            const now = new Date();
            const pad = n => String(n).padStart(2, '0');
            const formattedDate = `${now.getFullYear()}/${pad(now.getMonth() + 1)}/${pad(now.getDate())} ${pad(now.getHours())}:${pad(now.getMinutes())}`;

            const list = loadHighScores();
            list.push({ score: s, stage: st, date: formattedDate });
            list.sort((a, b) => b.score - a.score);
            const top10 = list.slice(0, 10);
            localStorage.setItem(HIGHSCORE_STORAGE_KEY, JSON.stringify(top10));
            return top10;
        } catch (_) {
            return loadHighScores();
        }
    }

    function clearHighScores() {
        try {
            localStorage.removeItem(HIGHSCORE_STORAGE_KEY);
        } catch (_) {}
    }

    // ==========================================
    // 1. サウンドエンジン (Web Audio API + Munchies!ボイス)
    // ==========================================
    class SoundEngine {
        constructor() {
            this.ctx = null;
            this.muted = false;
            this.bgmPlaying = false;
            this.bgmMode = 'normal'; // 'normal' or 'boss'
            this.bgmVariation = 0;  // 通常BGMバリエーション (0〜2)
            this.bossVariation = 0; // ボスBGMバリエーション (0〜1)
            this.bgmStep = 0;
            this.bgmTimer = null;
            this.nextNoteTime = 0;
        }

        init() {
            if (!this.ctx) {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                this.ctx = new AudioCtx();
            }
            if (this.ctx.state === 'suspended') {
                this.ctx.resume();
            }
        }

        toggleMute() {
            this.muted = !this.muted;
            if (this.muted) {
                this.stopBGM(false); // 状態は保持しつつ発音停止
            } else if (this.bgmPlaying) {
                this.startBGM(this.bgmMode, this.bgmVariation, this.bossVariation);
            }
            return this.muted;
        }

        // チップチューン風サイケデリックBGM開始（複数バリエーション対応）
        startBGM(mode = 'normal', normalVar = null, bossVar = null) {
            this.bgmMode = mode;
            if (normalVar !== null) this.bgmVariation = normalVar % 3;
            if (bossVar !== null) this.bossVariation = bossVar % 2;
            this.bgmPlaying = true;
            if (this.muted || !this.ctx) return;

            if (this.bgmTimer) {
                clearInterval(this.bgmTimer);
                this.bgmTimer = null;
            }

            this.bgmStep = 0;
            this.nextNoteTime = this.ctx.currentTime + 0.05;

            this.bgmTimer = setInterval(() => {
                if (!this.bgmPlaying || this.muted || !this.ctx) return;
                let stepDuration = 0.135;
                if (this.bgmMode === 'boss') {
                    stepDuration = this.bossVariation === 0 ? 0.115 : 0.105;
                } else {
                    const durations = [0.135, 0.125, 0.145];
                    stepDuration = durations[this.bgmVariation % durations.length];
                }
                while (this.nextNoteTime < this.ctx.currentTime + 0.12) {
                    this.scheduleBgmStep(this.bgmStep, this.nextNoteTime, stepDuration);
                    this.nextNoteTime += stepDuration;
                    this.bgmStep = (this.bgmStep + 1) % 32;
                }
            }, 35);
        }

        setBgmMode(mode, normalVar = null, bossVar = null) {
            const varChanged =
                (mode === 'normal' && normalVar !== null && normalVar !== this.bgmVariation) ||
                (mode === 'boss' && bossVar !== null && bossVar !== this.bgmVariation);
            if (this.bgmMode === mode && !varChanged) return;
            this.startBGM(mode, normalVar, bossVar);
        }

        stopBGM(resetFlag = true) {
            if (resetFlag) this.bgmPlaying = false;
            if (this.bgmTimer) {
                clearInterval(this.bgmTimer);
                this.bgmTimer = null;
            }
        }

        scheduleBgmStep(step, time, dur) {
            if (!this.ctx || this.muted) return;

            if (this.bgmMode === 'normal') {
                if (this.bgmVariation === 0) {

        playChipNote(freq, time, duration, type, volume, endFreq = null) {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = type;
            osc.frequency.setValueAtTime(freq, time);
            if (endFreq) {
                osc.frequency.linearRampToValueAtTime(endFreq, time + duration);
            }

            gain.gain.setValueAtTime(volume, time);
            gain.gain.exponentialRampToValueAtTime(0.0008, time + duration);

            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(time);
            osc.stop(time + duration);
        }

        playChipKick(time, volume) {
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = 'sine';
            osc.frequency.setValueAtTime(140, time);
            osc.frequency.exponentialRampToValueAtTime(38, time + 0.09);

            gain.gain.setValueAtTime(volume, time);
            gain.gain.exponentialRampToValueAtTime(0.001, time + 0.09);

            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(time);
            osc.stop(time + 0.09);
        }

        playChipHat(time, volume) {
            const bufferSize = Math.floor(this.ctx.sampleRate * 0.04);
            const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
            const data = buffer.getChannelData(0);
            for (let i = 0; i < bufferSize; i++) {
                data[i] = Math.random() * 2 - 1;
            }
            const noise = this.ctx.createBufferSource();
            noise.buffer = buffer;

            const filter = this.ctx.createBiquadFilter();
            filter.type = 'highpass';
            filter.frequency.setValueAtTime(6500, time);

            const gain = this.ctx.createGain();
            gain.gain.setValueAtTime(volume, time);
            gain.gain.exponentialRampToValueAtTime(0.0008, time + 0.04);

            noise.connect(filter);
            filter.connect(gain);
            gain.connect(this.ctx.destination);
            noise.start(time);
        }

        // 煙発射音（シュポッという柔らかいノイズ＋サイン波）
        playShoot(isRainbow = false) {
            if (this.muted || !this.ctx) return;
            const now = this.ctx.currentTime;
            const osc = this.ctx.createOscillator();
            const gain = this.ctx.createGain();
            osc.type = isRainbow ? 'triangle' : 'sine';
            const baseFreq = isRainbow ? 420 : 280;
            osc.frequency.setValueAtTime(baseFreq + Math.random() * 60, now);
            osc.frequency.exponentialRampToValueAtTime(90, now + 0.08);

            gain.gain.setValueAtTime(0.06, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);

            osc.connect(gain);
            gain.connect(this.ctx.destination);
            osc.start(now);
            osc.stop(now + 0.08);
        }

        // 敵撃破音（ポップな爆発音）
        playExplosion(isBoss = false) {
            if (this.muted || !this.ctx) return;
            const now = this.ctx.currentTime;
            const duration = isBoss ? 0.8 : 0.22;

            const bufferSize = this.ctx.sampleRate * duration;
            const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
            const data = buffer.getChannelData(0);
            for (let i = 0; i < bufferSize; i++) {
                data[i] = Math.random() * 2 - 1;
            }
            const noise = this.ctx.createBufferSource();
            noise.buffer = buffer;

            const filter = this.ctx.createBiquadFilter();
            filter.type = 'lowpass';
            filter.frequency.setValueAtTime(isBoss ? 600 : 900, now);
            filter.frequency.exponentialRampToValueAtTime(60, now + duration);

            const gain = this.ctx.createGain();
            gain.gain.setValueAtTime(isBoss ? 0.35 : 0.15, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + duration);

            noise.connect(filter);
            filter.connect(gain);
            gain.connect(this.ctx.destination);
            noise.start(now);
        }

        playPowerUp(isRare = false) {

        // ピザ取得時「Munchies!」ボイス＆ファンファーレ
        playMunchiesVoice() {
            if (this.muted) return;
            if (this.ctx) {
                if (this.ctx.state === 'suspended') {
                    this.ctx.resume().catch(() => {});
                }
                const now = this.ctx.currentTime;
                [523.25, 659.25, 783.99, 1046.50].forEach((freq, idx) => {
                    const osc = this.ctx.createOscillator();
                    const gain = this.ctx.createGain();
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(freq, now + idx * 0.05);
                    gain.gain.setValueAtTime(0.14, now + idx * 0.05);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.05 + 0.22);
                    osc.connect(gain);
                    gain.connect(this.ctx.destination);
                    osc.start(now + idx * 0.05);
                    osc.stop(now + idx * 0.05 + 0.22);
                });
            }

            let spokenBySynth = false;
            try {
                if ('speechSynthesis' in window) {
                    const synth = window.speechSynthesis;
                    const utter = new SpeechSynthesisUtterance('Munchies');
                    utter.lang = 'en-US';
                    utter.rate = 1.02;
                    utter.pitch = 1.18;
                    utter.volume = 1.0;

                    const voices = synth.getVoices ? synth.getVoices() : [];
                    if (Array.isArray(voices) && voices.length > 0) {
                        const preferred =
                            voices.find(v => /en/i.test(v.lang) && /natural|google|samantha|alex|daniel|aria|jenny/i.test(v.name)) ||
                            voices.find(v => /en/i.test(v.lang)) ||
                            voices[0];
                        if (preferred) {
                            utter.voice = preferred;
                            utter.lang = preferred.lang || 'en-US';
                        }
                    }

                    utter.onstart = () => {
                        spokenBySynth = true;
                    };

                    try {
                        synth.resume();
                    } catch (_) {}
                    synth.cancel();
                    synth.speak(utter);

                    // ブラウザによっては初回発話が遅延・キャンセルされるため短いフォールバックを用意
                    setTimeout(() => {
                        if (!this.muted && !spokenBySynth && !synth.speaking) {
                            this.playSynthMunchiesFallback();
                        }
                    }, 220);
                    return;
                }
            } catch (_) {}

            this.playSynthMunchiesFallback();
        }

        // SpeechSynthesisが使えない／無音環境向けの「Munchies」シンセボイス
        playSynthMunchiesFallback() {
            if (this.muted || !this.ctx) return;
            try {
                const now = this.ctx.currentTime;
                // "Mun-"（低め〜上がる母音）と "-chies"（高めで弾む声）の2音節トーン＋子音ノイズ
                const syllables = [
                    { start: 0.02, dur: 0.14, f0: 210, f1: 275, form: 540, vol: 0.18 },
                    { start: 0.17, dur: 0.22, f0: 310, f1: 410, form: 920, vol: 0.20 }
                ];

                syllables.forEach(s => {
                    const osc = this.ctx.createOscillator();
                    const sub = this.ctx.createOscillator();
                    const formant = this.ctx.createBiquadFilter();
                    const gain = this.ctx.createGain();

                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(s.f0, now + s.start);
                    osc.frequency.exponentialRampToValueAtTime(s.f1, now + s.start + s.dur);

                    sub.type = 'triangle';
                    sub.frequency.setValueAtTime(s.f0 * 1.5, now + s.start);
                    sub.frequency.exponentialRampToValueAtTime(s.f1 * 1.5, now + s.start + s.dur);

                    formant.type = 'bandpass';
                    formant.frequency.setValueAtTime(s.form, now + s.start);
                    formant.Q.setValueAtTime(3.2, now + s.start);

                    gain.gain.setValueAtTime(0.001, now + s.start);
                    gain.gain.linearRampToValueAtTime(s.vol, now + s.start + 0.025);
                    gain.gain.exponentialRampToValueAtTime(0.001, now + s.start + s.dur);

                    osc.connect(formant);
                    sub.connect(formant);
                    formant.connect(gain);
                    gain.connect(this.ctx.destination);

                    osc.start(now + s.start);
                    sub.start(now + s.start);
                    osc.stop(now + s.start + s.dur + 0.02);
                    sub.stop(now + s.start + s.dur + 0.02);
                });

                // "-ch-" の子音アタック用ショートノイズ
                const nLen = Math.floor(this.ctx.sampleRate * 0.06);
                const nBuf = this.ctx.createBuffer(1, nLen, this.ctx.sampleRate);
                const chData = nBuf.getChannelData(0);
                for (let i = 0; i < nLen; i++) {
                    chData[i] = (Math.random() * 2 - 1) * (1 - i / nLen);
                }
                const nSrc = this.ctx.createBufferSource();
                nSrc.buffer = nBuf;
                const hp = this.ctx.createBiquadFilter();
                hp.type = 'bandpass';
                hp.frequency.setValueAtTime(2600, now + 0.14);
                hp.Q.setValueAtTime(1.5, now + 0.14);
                const nGain = this.ctx.createGain();
                nGain.gain.setValueAtTime(0.14, now + 0.14);
                nGain.gain.exponentialRampToValueAtTime(0.001, now + 0.20);
                nSrc.connect(hp);
                hp.connect(nGain);
                nGain.connect(this.ctx.destination);
                nSrc.start(now + 0.14);
            } catch (_) {}
        }

        }

        }

        togglePause(forcePause = null) {
            if (this.state !== 'PLAYING' && this.state !== 'PAUSED') return;
            const nextPause = forcePause !== null ? Boolean(forcePause) : (this.state === 'PLAYING');

            if (nextPause) {
                this.state = 'PAUSED';
                this.pointer.active = false;
                this.keys = {};
                sound.stopBGM(true);
                this.pauseBtn.textContent = '▶';
                this.pauseScreen.classList.remove('hidden');
            } else {
                this.state = 'PLAYING';
                this.lastTime = performance.now();
                this.pauseBtn.textContent = '⏸';
                this.pauseScreen.classList.add('hidden');
                this.highScoreModal.classList.add('hidden');
                const mode = this.boss ? 'boss' : 'normal';
                sound.startBGM(mode, this.getNormalBgmVar(this.stage), this.getBossBgmVar(this.stage));
            }
        }

                const medal = idx === 0 ? '🥇' : idx === 1 ? '🥈' : idx === 2 ? '🥉' : `#${idx + 1}`;
                const rowHighlight = idx === 0 ? 'bg-amber-400/15 border-amber-400/40' : 'bg-white/5 border-white/10';
                return `
                    <div class="flex items-center justify-between gap-2 rounded-2xl border ${rowHighlight} px-3 py-2 text-xs">

        spawnParticles(x, y, color, count = 12, speed = 150) {
            for (let i = 0; i < count; i++) {
                const angle = Math.random() * Math.PI * 2;
                const spd = Math.random() * speed + 30;
                this.particles.push({
                    x, y,
                    vx: Math.cos(angle) * spd,
                    vy: Math.sin(angle) * spd,
                    radius: Math.random() * 5 + 2,
                    color,
                    life: Math.random() * 0.4 + 0.25,
                    maxLife: 0.65
                });
            }
        }

        // ==========================================
        // 4. メインループ & 更新処理
        // ==========================================
        loop(timestamp) {
            const dt = Math.min((timestamp - this.lastTime) / 1000, 0.05);
            this.lastTime = timestamp;

            if (this.state !== 'PAUSED') {
                this.updateBackground(dt);
            }

            if (this.state === 'PLAYING') {
                this.updatePlaying(dt);
            }

            this.draw();
            requestAnimationFrame((t) => this.loop(t));
        }

        updateBackground(dt) {
            this.bgHue = (this.bgHue + dt * 15) % 360;

            for (const c of this.tieDyeClouds) {
                c.y += c.speed * dt;
                c.hue = (c.hue + dt * 20) % 360;
                if (c.y - c.radius > this.height) {
                    c.y = -c.radius;
                    c.x = Math.random() * this.width;
                }
            }
            for (const s of this.stars) {
                s.y += s.speed * dt;
                if (s.y > this.height) {
                    s.y = 0;
                    s.x = Math.random() * this.width;
                }
            }
        }

        updatePlaying(dt) {
            if (this.screenShake > 0) this.screenShake = Math.max(0, this.screenShake - dt * 25);
            if (this.munchiesFlash > 0) this.munchiesFlash = Math.max(0, this.munchiesFlash - dt * 2.2);

            this.updatePlayer(dt);
            this.updateStageAndSpawns(dt);
            this.updateBullets(dt);
            this.updateEnemies(dt);
            this.updateBoss(dt);
            this.updateEnemyBullets(dt);
            this.updateItems(dt);
            this.updateParticlesAndTexts(dt);

            // タイマー系HUDの定期更新
            if (Math.random() < 0.2) this.updateHUD();
        }

        updatePlayer(dt) {
            const p = this.player;
            p.animPhase += dt * 8;

            if (p.invulnTimer > 0) p.invulnTimer -= dt;
            if (p.shieldTimer > 0) p.shieldTimer -= dt;
            if (p.mushroomTimer > 0) p.mushroomTimer -= dt;
            if (p.rainbowTimer > 0) p.rainbowTimer -= dt;

            // キーボード移動
            let dx = 0, dy = 0;
            if (this.keys['ArrowLeft'] || this.keys['KeyA']) dx -= 1;
            if (this.keys['ArrowRight'] || this.keys['KeyD']) dx += 1;
            if (this.keys['ArrowUp'] || this.keys['KeyW']) dy -= 1;
            if (this.keys['ArrowDown'] || this.keys['KeyS']) dy += 1;

            if (dx !== 0 || dy !== 0) {
                const len = Math.hypot(dx, dy);
                p.x += (dx / len) * p.speed * dt;
                p.y += (dy / len) * p.speed * dt;
            } else if (this.pointer.active) {
                const diffX = this.pointer.x - p.x;
                const diffY = this.pointer.y - p.y;
                p.x += diffX * Math.min(1, dt * 18);
                p.y += diffY * Math.min(1, dt * 18);
            }

            // 画面内制限
            p.x = Math.max(24, Math.min(this.width - 24, p.x));
            p.y = Math.max(70, Math.min(this.height - 32, p.y));

            // 自動煙発射
            p.shootTimer -= dt;
            if (p.shootTimer <= 0) {
                p.shootTimer = p.fireRate;
                this.firePlayerSmoke();
            }
        }

        firePlayerSmoke() {
            const p = this.player;
            const isRainbow = p.rainbowTimer > 0;
            const isGiant = p.mushroomTimer > 0;
            sound.playShoot(isRainbow);

            const baseDamage = (isRainbow ? 2.2 : 1.0) * (isGiant ? 1.8 : 1.0);
            const baseRadius = (isGiant ? 20 : 10) * (isRainbow ? 1.25 : 1.0);
            const mouthY = p.y - 26;

            const ways = p.wayCount;
            const spreadAngle = 0.14;
            for (let i = 0; i < ways; i++) {
                const offsetIdx = i - (ways - 1) / 2;
                const angle = -Math.PI / 2 + offsetIdx * spreadAngle;
                this.bullets.push({
                    x: p.x + offsetIdx * 6,
                    y: mouthY,
                    vx: Math.cos(angle) * p.bulletSpeed,
                    vy: Math.sin(angle) * p.bulletSpeed,
                    radius: baseRadius,
                    damage: baseDamage,
                    isRainbow,
                    isPierce: isGiant,
                    hitEnemies: new Set(),
                    hue: (performance.now() * 0.3 + i * 40) % 360,
                    wobble: Math.random() * Math.PI * 2
                });
            }

            if (p.bears >= 1) {
                this.bullets.push({
                    x: p.x - 34,
                    y: p.y + 4,
                    vx: 0,
                    vy: -p.bulletSpeed * 0.95,
                    radius: baseRadius * 0.75,
                    damage: baseDamage * 0.7,
                    isRainbow,
                    isPierce: isGiant,
                    hitEnemies: new Set(),
                    hue: 120,
                    wobble: 0
                });
            }
            if (p.bears >= 2) {
                this.bullets.push({
                    x: p.x + 34,
                    y: p.y + 4,
                    vx: 0,
                    vy: -p.bulletSpeed * 0.95,
                    radius: baseRadius * 0.75,
                    damage: baseDamage * 0.7,
                    isRainbow,
                    isPierce: isGiant,
                    hitEnemies: new Set(),
                    hue: 280,
                    wobble: Math.PI
                });
            }
        }

        updateStageAndSpawns(dt) {
            if (this.boss) return;

            if (this.bossWarningTimer > 0) {
                this.bossWarningTimer -= dt;
                if (this.bossWarningTimer <= 0) {
                    this.spawnBoss();
                }
                return;
            }

            this.stageTimer += dt;
            const stageDuration = 22 + Math.min(this.stage * 3, 15);

            if (this.stageTimer >= stageDuration) {
                this.bossWarningTimer = 2.4;
                sound.playBossWarning();
                this.addFloatingText(this.width / 2, this.height * 0.35, '⚠️ WARNING! BOSS APPROACHING ⚠️', '#ef4444', 20);
                return;
            }

            this.spawnTimer -= dt;
            const spawnInterval = Math.max(0.45, 1.35 - this.stage * 0.12);
            if (this.spawnTimer <= 0) {
                this.spawnTimer = spawnInterval;
                this.spawnEnemy();
            }
        }

        spawnEnemy() {
            const isGangster = Math.random() < (0.3 + this.stage * 0.05);
            const margin = 38;
            const x = margin + Math.random() * (this.width - margin * 2);
            const hpScale = 1 + (this.stage - 1) * 0.35;

            if (!isGangster) {
                this.enemies.push({
                    type: 'police',
                    x,
                    y: -30,
                    vx: (Math.random() < 0.5 ? -1 : 1) * (40 + Math.random() * 40),
                    vy: 95 + this.stage * 10,
                    radius: 20,
                    hp: Math.ceil(3 * hpScale),
                    maxHp: Math.ceil(3 * hpScale),
                    shootTimer: 1.2 + Math.random() * 1.2,
                    score: 150,
                    phase: Math.random() * Math.PI * 2
                });
            } else {
                this.enemies.push({
                    type: 'gangster',
                    x,
                    y: -35,
                    vx: (Math.random() < 0.5 ? -1 : 1) * (75 + this.stage * 8),
                    vy: 55 + this.stage * 5,
                    radius: 23,
                    hp: Math.ceil(6 * hpScale),
                    maxHp: Math.ceil(6 * hpScale),
                    shootTimer: 1.0 + Math.random() * 1.0,
                    score: 300,
                    phase: Math.random() * Math.PI * 2
                });
            }
        }

        spawnBoss() {
            const bossTypes = [
                { name: 'CHIEF PATROL (巨大パトカーメカ)', kind: 'patrol', color: '#3b82f6' },
                { name: 'BIG BOSS GANGSTA (マフィアのドン)', kind: 'mafia', color: '#a855f7' },
                { name: 'PSYCHEDELIC CHOPPER (サイケヘリ)', kind: 'chopper', color: '#ec4899' }
            ];
            const template = bossTypes[(this.stage - 1) % bossTypes.length];
            const maxHp = Math.floor(110 * Math.pow(1.38, this.stage - 1));

            this.boss = {
                name: `ST.${this.stage} ${template.name}`,
                kind: template.kind,
                color: template.color,
                x: this.width / 2,
                y: -80,
                targetY: 145,
                vx: 85 + this.stage * 10,
                radius: 54,
                hp: maxHp,
                maxHp: maxHp,
                attackTimer: 1.5,
                patternIdx: 0,
                phase: 0
            };

            this.bossHpContainer.classList.remove('hidden');
            this.bossNameEl.textContent = this.boss.name;
            sound.setBgmMode('boss', this.getNormalBgmVar(this.stage), this.getBossBgmVar(this.stage));
            this.updateBossHpUI();
        }

        updateBossHpUI() {
            if (!this.boss) return;
            const pct = Math.max(0, (this.boss.hp / this.boss.maxHp) * 100);
            this.bossHpBar.style.width = `${pct}%`;
            this.bossHpText.textContent = `${Math.ceil(pct)}%`;
        }

        updateBullets(dt) {
            for (let i = this.bullets.length - 1; i >= 0; i--) {
                const b = this.bullets[i];
                b.wobble += dt * 12;
                b.x += b.vx * dt + Math.sin(b.wobble) * 18 * dt;
                b.y += b.vy * dt;
                b.hue = (b.hue + dt * 360) % 360;

                if (b.y + b.radius < -20 || b.x < -30 || b.x > this.width + 30) {
                    this.bullets.splice(i, 1);
                }
            }
        }

        updateEnemies(dt) {
            for (let i = this.enemies.length - 1; i >= 0; i--) {
                const e = this.enemies[i];
                e.phase += dt * 4;

                if (e.type === 'police') {
                    e.x += Math.sin(e.phase) * e.vx * dt;
                    e.y += e.vy * dt;
                } else {
                    e.x += e.vx * dt;
                    e.y += e.vy * dt;
                    if (e.x < 30 || e.x > this.width - 30) e.vx *= -1;
                }

                e.shootTimer -= dt;
                if (e.shootTimer <= 0 && e.y > 20 && e.y < this.height - 140) {
                    e.shootTimer = Math.max(1.1, 2.3 - this.stage * 0.15);
                    this.fireEnemyBullet(e);
                }

                let destroyed = false;
                for (let j = this.bullets.length - 1; j >= 0; j--) {
                    const b = this.bullets[j];
                    if (b.isPierce && b.hitEnemies.has(e)) continue;

                    const dist = Math.hypot(e.x - b.x, e.y - b.y);
                    if (dist < e.radius + b.radius) {
                        e.hp -= b.damage;
                        this.spawnParticles(b.x, b.y, b.isRainbow ? `hsl(${b.hue},100%,65%)` : '#e2e8f0', 4, 70);

                        if (b.isPierce) {
                            b.hitEnemies.add(e);
                        } else {
                            this.bullets.splice(j, 1);
                        }

                        if (e.hp <= 0) {
                            this.destroyEnemy(e);
                            this.enemies.splice(i, 1);
                            destroyed = true;
                            break;
                        }
                    }
                }
                if (destroyed) continue;

                const pDist = Math.hypot(e.x - this.player.x, e.y - this.player.y);
                if (pDist < e.radius + this.player.radius - 4) {
                    this.damagePlayer();
                    this.destroyEnemy(e);
                    this.enemies.splice(i, 1);
                    continue;
                }

                if (e.y - e.radius > this.height + 30) {
                    this.enemies.splice(i, 1);
                }
            }
        }

        fireEnemyBullet(e) {
            const angleToPlayer = Math.atan2(this.player.y - e.y, this.player.x - e.x);
            const speed = 190 + this.stage * 18;

            if (e.type === 'police') {
                this.enemyBullets.push({
                    x: e.x,
                    y: e.y + 15,
                    vx: Math.cos(angleToPlayer) * speed,
                    vy: Math.sin(angleToPlayer) * speed,
                    radius: 6,
                    color: '#38bdf8'
                });
            } else {
                [-0.16, 0.16].forEach(offset => {
                    this.enemyBullets.push({
                        x: e.x,
                        y: e.y + 15,
                        vx: Math.cos(angleToPlayer + offset) * (speed * 1.05),
                        vy: Math.sin(angleToPlayer + offset) * (speed * 1.05),
                        radius: 7,
                        color: '#f43f5e'
                    });
                });
            }
        }

        destroyEnemy(e) {
            sound.playExplosion(false);
            this.score += e.score;
            this.updateHUD();
            this.spawnParticles(e.x, e.y, e.type === 'police' ? '#60a5fa' : '#c084fc', 14, 140);

            if (Math.random() < 0.38) {
                this.spawnItem(e.x, e.y);
            }
        }

        updateBoss(dt) {
            const b = this.boss;
            if (!b) return;

            b.phase += dt * 3;
            if (b.y < b.targetY) {
                b.y += 70 * dt;
            } else {
                b.x += b.vx * dt;
                if (b.x < b.radius + 20 || b.x > this.width - b.radius - 20) {
                    b.vx *= -1;
                }
            }

            b.attackTimer -= dt;
            if (b.attackTimer <= 0 && b.y >= b.targetY - 5) {
                b.patternIdx = (b.patternIdx + 1) % 3;
                const speed = 210 + this.stage * 15;

                if (b.patternIdx === 0) {
                    b.attackTimer = Math.max(0.9, 1.5 - this.stage * 0.08);
                    const ways = 5 + Math.min(this.stage, 4);
                    for (let i = 0; i < ways; i++) {
                        const angle = Math.PI / 2 + (i - (ways - 1) / 2) * 0.22;
                        this.enemyBullets.push({
                            x: b.x,
                            y: b.y + 35,
                            vx: Math.cos(angle) * speed,
                            vy: Math.sin(angle) * speed,
                            radius: 8,
                            color: '#fb7185'
                        });
                    }
                } else if (b.patternIdx === 1) {
                    b.attackTimer = Math.max(0.8, 1.3 - this.stage * 0.06);
                    const baseAngle = Math.atan2(this.player.y - b.y, this.player.x - b.x);
                    [-0.12, 0, 0.12].forEach(off => {
                        this.enemyBullets.push({
                            x: b.x,
                            y: b.y + 30,
                            vx: Math.cos(baseAngle + off) * (speed * 1.15),
                            vy: Math.sin(baseAngle + off) * (speed * 1.15),
                            radius: 7,
                            color: '#facc15'
                        });
                    });
                } else {
                    b.attackTimer = Math.max(1.1, 1.7 - this.stage * 0.08);
                    const count = 10 + Math.min(this.stage * 2, 8);
                    for (let i = 0; i < count; i++) {
                        const angle = (Math.PI * 2 / count) * i + b.phase;
                        this.enemyBullets.push({
                            x: b.x,
                            y: b.y,
                            vx: Math.cos(angle) * (speed * 0.85),
                            vy: Math.sin(angle) * (speed * 0.85),
                            radius: 7,
                            color: '#38bdf8'
                        });
                    }
                }
            }

            for (let j = this.bullets.length - 1; j >= 0; j--) {
                const bul = this.bullets[j];
                if (bul.isPierce && bul.hitEnemies.has(b)) continue;

                const dist = Math.hypot(b.x - bul.x, b.y - bul.y);
                if (dist < b.radius + bul.radius) {
                    b.hp -= bul.damage;
                    this.spawnParticles(bul.x, bul.y, '#ffffff', 4, 80);

                    if (bul.isPierce) {
                        bul.hitEnemies.add(b);
                    } else {
                        this.bullets.splice(j, 1);
                    }

                    this.updateBossHpUI();

                    if (b.hp <= 0) {
                        this.defeatBoss();
                        return;
                    }
                }
            }

            const pDist = Math.hypot(b.x - this.player.x, b.y - this.player.y);
            if (pDist < b.radius + this.player.radius - 6) {
                this.damagePlayer();
            }
        }

        defeatBoss() {
            const b = this.boss;
            sound.playExplosion(true);
            this.screenShake = 18;
            this.enemyBullets = [];

            const bonus = 2000 * this.stage;
            this.score += bonus;
            this.spawnParticles(b.x, b.y, '#facc15', 45, 240);
            this.spawnParticles(b.x, b.y, '#f472b6', 35, 200);
            this.addFloatingText(b.x, b.y, `BOSS PURIFIED! +${bonus}`, '#fde047', 22);

            this.spawnItem(b.x - 30, b.y);
            this.spawnItem(b.x + 30, b.y);
            if (this.stage % 2 === 0) this.spawnItem(b.x, b.y - 20);

            this.boss = null;
            this.bossHpContainer.classList.add('hidden');

            this.stage++;
            this.stageTimer = 0;
            sound.setBgmMode('normal', this.getNormalBgmVar(this.stage), this.getBossBgmVar(this.stage));
            this.updateHUD();

            setTimeout(() => {
                if (this.state === 'PLAYING') {
                    this.addFloatingText(this.width / 2, this.height * 0.4, `STAGE ${this.stage} START!`, '#38bdf8', 26);
                }
            }, 1200);
        }

        updateEnemyBullets(dt) {
            for (let i = this.enemyBullets.length - 1; i >= 0; i--) {
                const eb = this.enemyBullets[i];
                eb.x += eb.vx * dt;
                eb.y += eb.vy * dt;

                const dist = Math.hypot(eb.x - this.player.x, eb.y - this.player.y);
                if (dist < eb.radius + this.player.radius - 7) {
                    this.enemyBullets.splice(i, 1);
                    this.damagePlayer();
                    continue;
                }

                if (eb.y > this.height + 20 || eb.y < -20 || eb.x < -20 || eb.x > this.width + 20) {
                    this.enemyBullets.splice(i, 1);
                }
            }
        }

        damagePlayer() {
            const p = this.player;
            if (p.invulnTimer > 0 || p.shieldTimer > 0) return;

            sound.playHit();
            this.screenShake = 14;
            p.life--;
            p.invulnTimer = 2.2;

            if (p.wayCount > 1) p.wayCount--;
            if (p.bears > 0) p.bears--;

            this.spawnParticles(p.x, p.y, '#ef4444', 22, 160);
            this.addFloatingText(p.x, p.y - 20, 'OUCH!!', '#f87171', 20);
            this.updateHUD();

            if (p.life <= 0) {
                this.gameOver();
            }
        }

        spawnItem(x, y) {
            const totalWeight = ITEM_TYPES.reduce((sum, item) => sum + item.weight, 0);
            let r = Math.random() * totalWeight;
            let selected = ITEM_TYPES[0];
            for (const item of ITEM_TYPES) {
                if (r < item.weight) {
                    selected = item;
                    break;
                }
                r -= item.weight;
            }

            this.items.push({
                ...selected,
                x,
                y,
                vy: 85,
                radius: 18,
                wobble: Math.random() * Math.PI * 2
            });
        }

        updateItems(dt) {
            for (let i = this.items.length - 1; i >= 0; i--) {
                const it = this.items[i];
                it.wobble += dt * 4;
                it.x += Math.sin(it.wobble) * 35 * dt;
                it.y += it.vy * dt;

                const dist = Math.hypot(it.x - this.player.x, it.y - this.player.y);
                if (dist < it.radius + this.player.radius + 10) {
                    this.applyItemEffect(it);
                    this.items.splice(i, 1);
                    continue;
                }

                if (it.y > this.height + 30) {
                    this.items.splice(i, 1);
                }
            }
        }

        applyItemEffect(item) {
            const p = this.player;
            this.score += 200;
            this.spawnParticles(item.x, item.y, item.color, 16, 130);

            if (item.id === 'pizza') {
                sound.playMunchiesVoice();
                this.munchiesFlash = 1.0;
                this.screenShake = 12;
                this.score += 1000;
                this.enemyBullets = [];
                this.addFloatingText(p.x, p.y - 38, '🍕 Munchies!!', '#facc15', 22);

                for (let i = this.enemies.length - 1; i >= 0; i--) {
                    this.destroyEnemy(this.enemies[i]);
                }
                this.enemies = [];
                if (this.boss) {
                    this.boss.hp -= Math.max(40, this.boss.maxHp * 0.22);
                    this.updateBossHpUI();
                    if (this.boss.hp <= 0) this.defeatBoss();
                }
            } else {
                sound.playPowerUp(item.id === 'skull' || item.id === 'rainbow');
                this.addFloatingText(p.x, p.y - 35, item.label, item.color, 17);

                switch (item.id) {
                    case 'leaf':
                        p.wayCount = Math.min(5, p.wayCount + 1);
                        break;
                    case 'joint':
                        p.fireRate = Math.max(0.09, p.fireRate * 0.84);
                        p.bulletSpeed = Math.min(720, p.bulletSpeed + 35);
                        break;
                    case 'cookie':
                        p.speed = Math.min(480, p.speed + 25);
                        this.score += 500;
                        break;
                    case 'mushroom':
                        p.mushroomTimer = 9.0;
                        break;
                    case 'rainbow':
                        p.rainbowTimer = 10.0;
                        break;
                    case 'sunglasses':
                        p.shieldTimer = 9.0;
                        break;
                    case 'bear':
                        if (p.bears < 2) {
                            p.bearColors[p.bears] = BEAR_COLORS[Math.floor(Math.random() * BEAR_COLORS.length)];
                            p.bears++;
                        } else {
                            this.score += 800;
                        }
                        break;
                    case 'skull':
                        p.life = Math.min(p.maxLife, p.life + 1);
                        break;
                }
            }
            this.updateHUD();
        }

        updateParticlesAndTexts(dt) {
            for (let i = this.particles.length - 1; i >= 0; i--) {
                const pt = this.particles[i];
                pt.x += pt.vx * dt;
                pt.y += pt.vy * dt;
                pt.life -= dt;
                if (pt.life <= 0) this.particles.splice(i, 1);
            }
            for (let i = this.floatingTexts.length - 1; i >= 0; i--) {
                const ft = this.floatingTexts[i];
                ft.y -= 35 * dt;
                ft.life -= dt;
                if (ft.life <= 0) this.floatingTexts.splice(i, 1);
            }
        }

        // ==========================================
        // 5. Canvas リッチ描画処理
        // ==========================================
        draw() {
            const ctx = this.ctx;
            ctx.save();

            if (this.screenShake > 0) {
                const sx = (Math.random() - 0.5) * this.screenShake;
                const sy = (Math.random() - 0.5) * this.screenShake;
                ctx.translate(sx, sy);
            }

            // 1. サイケデリック背景描画
            this.drawBackground(ctx);

            if (this.state === 'PLAYING' || this.state === 'PAUSED' || this.state === 'GAMEOVER') {
                // 2. 煙の弾描画
                this.drawSmokeBullets(ctx);

                // 3. アイテム描画
                this.drawItems(ctx);

                // 4. ザコ敵描画
                this.drawEnemies(ctx);

                // 5. ボス描画
                if (this.boss) this.drawBoss(ctx);

                // 6. 主人公「ボングくん」＆デッドベア描画
                if (this.state === 'PLAYING' || this.state === 'PAUSED') this.drawPlayer(ctx);

                // 7. 敵弾描画
                this.drawEnemyBullets(ctx);

                // 8. パーティクル＆ポップアップテキスト描画
                this.drawParticlesAndTexts(ctx);

                // ピザ（Munchies!）発動時のフラッシュ演出
                if (this.munchiesFlash > 0) {
                    ctx.fillStyle = `rgba(254, 240, 138, ${this.munchiesFlash * 0.55})`;
                    ctx.fillRect(0, 0, this.width, this.height);
                }
            }

            ctx.restore();
        }

        drawBackground(ctx) {
            const grad = ctx.createLinearGradient(0, 0, 0, this.height);
            grad.addColorStop(0, `hsl(${(this.bgHue + 260) % 360}, 65%, 10%)`);
            grad.addColorStop(0.5, `hsl(${(this.bgHue + 290) % 360}, 60%, 14%)`);
            grad.addColorStop(1, `hsl(${(this.bgHue + 320) % 360}, 70%, 9%)`);
            ctx.fillStyle = grad;
            ctx.fillRect(0, 0, this.width, this.height);

            for (const c of this.tieDyeClouds) {
                const rGrad = ctx.createRadialGradient(c.x, c.y, 5, c.x, c.y, c.radius);
                rGrad.addColorStop(0, `hsla(${c.hue}, 90%, 55%, 0.14)`);
                rGrad.addColorStop(0.6, `hsla(${(c.hue + 60) % 360}, 85%, 45%, 0.06)`);
                rGrad.addColorStop(1, 'rgba(0,0,0,0)');
                ctx.fillStyle = rGrad;
                ctx.beginPath();
                ctx.arc(c.x, c.y, c.radius, 0, Math.PI * 2);
                ctx.fill();
            }

            for (const s of this.stars) {
                ctx.fillStyle = `hsla(${(s.hue + this.bgHue) % 360}, 100%, 80%, 0.75)`;
                ctx.beginPath();
                ctx.arc(s.x, s.y, s.size, 0, Math.PI * 2);
                ctx.fill();
            }
        }

        drawBoss(ctx) {
            const b = this.boss;
            ctx.save();
            ctx.translate(b.x, b.y);

            const pulse = Math.sin(b.phase * 2) * 6;
            const aura = ctx.createRadialGradient(0, 0, 20, 0, 0, b.radius + 25 + pulse);
            aura.addColorStop(0, 'rgba(239, 68, 68, 0.45)');
            aura.addColorStop(1, 'rgba(0, 0, 0, 0)');
            ctx.fillStyle = aura;
            ctx.beginPath();
            ctx.arc(0, 0, b.radius + 25 + pulse, 0, Math.PI * 2);
            ctx.fill();

            if (b.kind === 'patrol') {
                ctx.fillStyle = '#0f172a';
                ctx.beginPath();
                ctx.roundRect(-52, -32, 104, 68, 16);
                ctx.fill();
                ctx.fillStyle = '#ffffff';
                ctx.fillRect(-48, -10, 96, 24);
                const flash = Math.floor(b.phase * 4) % 2 === 0;
                ctx.fillStyle = flash ? '#ef4444' : '#3b82f6';
                ctx.beginPath();
                ctx.roundRect(-24, -42, 20, 12, 4);
                ctx.fill();
                ctx.fillStyle = flash ? '#3b82f6' : '#ef4444';
                ctx.beginPath();
                ctx.roundRect(4, -42, 20, 12, 4);
                ctx.fill();

                ctx.fillStyle = '#1e1b4b';

                ctx.fill();

        drawEnemyBullets(ctx) {
            for (const eb of this.enemyBullets) {
                ctx.save();
                ctx.fillStyle = eb.color;
                ctx.shadowColor = eb.color;
                ctx.shadowBlur = 8;
                ctx.beginPath();
                ctx.arc(eb.x, eb.y, eb.radius, 0, Math.PI * 2);
                ctx.fill();

                ctx.fillStyle = '#ffffff';
                ctx.beginPath();
                ctx.arc(eb.x, eb.y, eb.radius * 0.45, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }
        }

        drawItems(ctx) {
            ctx.textAlign = 'center';
            ctx.textBaseline = 'middle';

            for (const it of this.items) {
                ctx.save();
                ctx.translate(it.x, it.y);

                ctx.fillStyle = 'rgba(15, 7, 32, 0.75)';
                ctx.strokeStyle = it.color;
                ctx.lineWidth = 2.5;
                ctx.shadowColor = it.color;
                ctx.shadowBlur = 12;
                ctx.beginPath();
                ctx.arc(0, 0, it.radius, 0, Math.PI * 2);
                ctx.fill();
                ctx.stroke();

                ctx.shadowBlur = 0;
                ctx.font = it.id === 'skull' ? '18px sans-serif' : '22px sans-serif';
                ctx.fillText(it.emoji, 0, 1);

                ctx.restore();
            }
        }

        drawParticlesAndTexts(ctx) {
            for (const pt of this.particles) {
                ctx.save();
                ctx.globalAlpha = Math.max(0, pt.life / pt.maxLife);
                ctx.fillStyle = pt.color;
                ctx.beginPath();
                ctx.arc(pt.x, pt.y, pt.radius, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
            }

            ctx.textAlign = 'center';
            for (const ft of this.floatingTexts) {
                ctx.save();
                ctx.globalAlpha = Math.min(1, (ft.life / ft.maxLife) * 1.5);
                ctx.font = `900 ${ft.size}px 'Fredoka', 'Noto Sans JP', sans-serif`;
                ctx.strokeStyle = '#090314';
                ctx.lineWidth = 4;
                ctx.strokeText(ft.text, ft.x, ft.y);
                ctx.fillStyle = ft.color;
                ctx.fillText(ft.text, ft.x, ft.y);
                ctx.restore();
            }
        }
    }
